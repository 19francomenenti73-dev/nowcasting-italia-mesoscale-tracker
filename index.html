import json import numpy as np from scipy.ndimage import label, center_of_mass def process_radar_data(dbz_matrix, lat_grid, lon_grid, speed_kmh, height_km): """ Elabora la matrice radar dBZ per estrarre il centroide reale, il vettore reale (passato) e il vettore predittivo (futuro). """ # 1. Soglia di riflettività per celle convettive a mesoscala (es. >= 42 dBZ) convective_mask = dbz_matrix >= 42.0 if not np.any(convective_mask): return None # Nessuna struttura severa rilevata # 2. Etichettatura delle componenti connesse (distinzione delle celle) labeled_array, num_features = label(convective_mask) macro_structures = [] for i in range(1, num_features + 1): cell_mask = (labeled_array == i) # Ignora strutture troppo piccole (filtro rumore) if np.sum(cell_mask) < 10: continue # 3. Calcolo del baricentro ponderato (massima intensità di riflettività) # Usiamo i valori dBZ come pesi per posizionare il centroide esattamente sul cuore della cella weights = dbz_matrix * cell_mask y_idx, x_idx = center_of_mass(weights) # Conversione indici matrice -> Coordinate Geografiche (Lat, Lon) # (Assumendo griglie 2D di latitudine e longitudine corrispondenti alla matrice radar) center_lat = float(np.interp(y_idx, np.arange(lat_grid.shape[0]), lat_grid[:, 0])) center_lon = float(np.interp(x_idx, np.arange(lon_grid.shape[1]), lon_grid[0, :])) # 4. Calcolo del raggio approssimato della struttura (in km) pixel_count = np.sum(cell_mask) radius_km = float(np.sqrt(pixel_count) * 2.5) # Fattore di scala risoluzione radar (es. 2.5 km/pixel) # 5. Generazione Vettore Reale (ultimi 15 minuti di storico posizioni) # In un sistema reale, salvi le posizioni precedenti in DB o cache. # Qui simuliamo lo storico a ritroso basandoci sulla direzione di moto opposta. actual_path = [ [center_lat - 0.05, center_lon - 0.05], [center_lat - 0.02, center_lon - 0.02], [center_lat, center_lon] # Punto corrente (deve coincidere col centroide) ] # 6. Generazione Vettore Predittivo (proiezione futura basata su velocità e direzione) # Semplificazione vettoriale basata su direzione stimata (es. verso est-nordest) # In un ambiente di produzione usi il vettore di moto da cross-correlation o Extrapolation Nowcasting (PySTEPS) forecast_path = [ [center_lat, center_lon], [center_lat + 0.04, center_lon + 0.05], [center_lat + 0.08, center_lon + 0.10] ] structure_data = { "id": f"MCS-{i:03d}", "type": "Sistema Convettivo a Mesoscala", "center": [center_lat, center_lon], # IL CENTROIDE FISSO SULLA CELLA "radius_km": round(radius_km, 1), "speed_kmh": int(speed_kmh), "height_km": float(height_km), "intensity": "Fase Temporale Forte", "distance_rank": "closest" if i == 1 else "intermediate", "actual_path": actual_path, "forecast_path": forecast_path, "cep_radius_km": 0.8 } macro_structures.append(structure_data) # Struttura del JSON finale che alimenta il frontend Leaflet output_json = { "radar_tile": { "host": "https://radar.protezionecivile.gov.it", "path": "/mosaic" }, "macro_structures": macro_structures } return output_json # Esempio di esportazione su file # with open('centroids.json', 'w') as f: # json.dump(output_json, f, indent=2)
